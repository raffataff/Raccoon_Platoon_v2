# Intel Console Interaction System - Implementation Plan

## Overview
Add intel console objects that raccoons can interact with by pressing 'E' when nearby. Interacting starts a hack timer, after which the console is captured as an objective. Enemies may spawn periodically during the hack.

## User Clarifications
- **Count per mission**: Varies by phase (1-3 based on phase scaling)
- **Hack behavior**: Hands-free - hack continues automatically even if raccoon walks away
- **Enemy spawn timing**: Periodic spawning throughout the hack duration (not just at start)

## 3. IntelConsole Class (`js/IntelConsole.js`)

### 3.1 Class Structure
```javascript
class IntelConsole {
    constructor(x, y, game, id) {
        this.x = x;
        this.y = y;
        this.game = game;
        this.id = id;
        this.type = 'intel_console';
        
        // State - HANDS-FREE: hack continues even if raccoon leaves
        this.isHacked = false;
        this.isBeingHacked = false;
        this.hackTimer = 0;
        this.hackDuration = 0;
        this.hackingRaccoon = null;
        this.spawnTimer = 0;
        this.spawnInterval = 3.0; // Seconds between enemy spawns during hack
        
        // Sprites
        this.imageOn = game.preloadedImages['assets/images/objects/possums/intel/intel_console_1_on.png'];
        this.imageOff = game.preloadedImages['assets/images/objects/possums/intel/intel_console_1_off.png'];
        
        // Dimensions
        this.width = this.imageOn.naturalWidth * 0.8;
        this.height = this.imageOn.naturalHeight * 0.8;
        
        // Interaction
        this.interactionRadius = 100;
        this.proximityText = "Press E to Hack";
    }
    
    update(deltaTime) {
        if (this.isBeingHacked && !this.isHacked) {
            this.hackTimer -= deltaTime;
            
            // PERIODIC ENEMY SPAWNING during hack
            this.spawnTimer -= deltaTime;
            if (this.spawnTimer <= 0) {
                this.spawnEnemiesDuringHack();
                const phase = this.game.currentPhaseIndex;
                this.spawnTimer = Math.max(1.0, this.spawnInterval - (phase * 0.2));
            }
            
            if (this.hackTimer <= 0) {
                this.completeHack();
            }
        }
    }
    
    render(ctx) {
        const sprite = this.isHacked ? this.imageOff : this.imageOn;
        ctx.drawImage(sprite, this.x, this.y, this.width, this.height);
        
        // Draw hack progress bar if being hacked
        if (this.isBeingHacked && !this.isHacked) {
            const progress = 1 - (this.hackTimer / this.hackDuration);
            const barWidth = 60;
            const barHeight = 8;
            const barX = this.x + (this.width - barWidth) / 2;
            const barY = this.y - 20;
            
            ctx.fillStyle = '#333';
            ctx.fillRect(barX, barY, barWidth, barHeight);
            ctx.fillStyle = '#00FF00';
            ctx.fillRect(barX, barY, barWidth * progress, barHeight);
        }
    }
    
    startHack(raccoon, duration) {
        this.isBeingHacked = true;
        this.hackingRaccoon = raccoon;
        this.hackTimer = duration;
        this.hackDuration = duration;
        this.spawnTimer = this.spawnInterval;
    }
    
    completeHack() {
        this.isHacked = true;
        this.isBeingHacked = false;
        this.hackingRaccoon = null;
        // Switch to "off" sprite (render handles this via isHacked flag)
    }
    
    spawnEnemiesDuringHack() {
        if (!this.game) return;
        
        const phase = this.game.currentPhaseIndex;
        const spawnChance = this.getSpawnChance(phase);
        
        if (Math.random() < spawnChance) {
            const countRange = this.getSpawnCount(phase);
            const spawnCount = Math.floor(countRange[0] + Math.random() * (countRange[1] - countRange[0]));
            
            for (let i = 0; i < spawnCount; i++) {
                const unitType = this.getRandomUnitType(phase);
                const spawnPos = this.getRandomSpawnPosition();
                // Spawn enemy at 300px from console, facing toward console
                this.game.spawnEnemyAtLocation(spawnPos.x, spawnPos.y, unitType, this.x, this.y);
            }
        }
    }
    
    getRandomSpawnPosition() {
        // Spawn at exactly 300px from the console in any direction (full 360 degrees)
        const angle = Math.random() * Math.PI * 2; // Random angle 0 to 2*PI (any direction)
        const radius = 300;
        return {
            x: this.x + Math.cos(angle) * radius,
            y: this.y + Math.sin(angle) * radius
        };
    }
    
    getSpawnChance(phase) {
        return Math.min(0.3 + (phase * 0.1), 0.8);
    }
    
    getSpawnCount(phase) {
        return {
            min: Math.min(1 + Math.floor(phase * 0.5), 5),
            max: Math.min(2 + Math.floor(phase * 1), 8)
        };
    }
    
    getRandomUnitType(phase) {
        const pool = CONFIG.BASE_PARAMETERS.intelSpawnUnitPool;
        const available = pool.filter(u => !u.unlocksPhase || u.unlocksPhase <= phase);
        // Weighted random selection
        const totalWeight = available.reduce((sum, u) => sum + u.weight, 0);
        let rand = Math.random() * totalWeight;
        for (const unit of available) {
            rand -= unit.weight;
            if (rand <= 0) return unit.type;
        }
        return 'possum_grunt';
    }
    
    getNearestRaccoonInRange() {
        if (!this.game || !this.game.deployedSquadRoster) return null;
        
        let nearest = null;
        let nearestDist = Infinity;
        
        for (const raccoon of this.game.deployedSquadRoster) {
            if (!raccoon.isAlive()) continue;
            const dist = distance(this.x, this.y, raccoon.x, raccoon.y);
            if (dist <= this.interactionRadius && dist < nearestDist) {
                nearest = raccoon;
                nearestDist = dist;
            }
        }
        return nearest;
    }
}
```

// New method:
handleIntelConsoleInteraction() {
    for (const console of this.intelConsoles) {
        if (!console.isHacked && !console.isBeingHacked && console.getNearestRaccoonInRange()) {
            this.startHackOnConsole(console);
            break;
        }
    }
}

// New method:
startHackOnConsole(console) {
    const phase = this.currentPhaseIndex;
    const baseTime = 10 + (phase * 2); // Base 10-20 seconds
    const bonusTime = 2 + (phase * 2); // +2-4 seconds per phase
    const duration = baseTime + Math.random() * bonusTime;
    
    console.startHack(this.selectedUnits[0], duration);
    
    // Maybe spawn enemies
    if (Math.random() < this.getIntelSpawnChance(phase)) {
        this.spawnEnemiesNearIntelConsole(console, phase);
    }
}

// New method:
spawnEnemiesNearIntelConsole(console, phase) {
    const countRange = this.getIntelSpawnCount(phase);
    const spawnCount = Math.floor(countRange[0] + Math.random() * (countRange[1] - countRange[0]));
    const spawnRadius = 300; // Fixed 300px from console
    
    for (let i = 0; i < spawnCount; i++) {
        const angle = Math.random() * Math.PI * 2;
        const dist = spawnRadius;
        const spawnX = console.x + Math.cos(angle) * dist;
        const spawnY = console.y + Math.sin(angle) * dist;
        
        // Spawn based on phase-unlocked pool, targeting console position
        const unitType = this.getRandomIntelSpawnUnit(phase);
        this.spawnEnemyAtLocation(spawnX, spawnY, unitType, console.x, console.y);
    }
}

// New method:
spawnEnemyAtLocation(x, y, unitType, targetX, targetY) {
    let newEnemy;
    switch(unitType) {
        case 'possum_grunt':
            newEnemy = new PossumGrunt(x, y, this.game, `PSM-INTEL-${this.enemyUnits.length + 1}`);
            break;
        case 'possum_heavy':
            newEnemy = new PossumHeavy(x, y, this.game, `PSM-INTEL-${this.enemyUnits.length + 1}`);
            break;
        case 'possum_sniper':
            newEnemy = new PossumSniper(x, y, this.game, `PSM-INTEL-${this.enemyUnits.length + 1}`);
            break;
        case 'possum_elite':
            newEnemy = new PossumElite(x, y, this.game, `PSM-INTEL-${this.enemyUnits.length + 1}`);
            break;
        default:
            newEnemy = new PossumGrunt(x, y, this.game, `PSM-INTEL-${this.enemyUnits.length + 1}`);
    }
    
    // Set enemy to move toward the intel console
    if (targetX !== undefined && targetY !== undefined) {
        newEnemy.setMoveTarget(targetX, targetY);
    }
    
    this.enemyUnits.push(newEnemy);
    if (this.spatialGrid) {
        this.spatialGrid.addObject(newEnemy);
    }
    if (typeof this.incrementObjectiveEnemyCount === 'function') {

    }
}

## 10. Edge Cases

| Scenario | Handling |
|----------|----------|
| All raccoons die during hack | Hack continues (hands-free), but completion requires raccoons to still be alive at mission end |
| Multiple raccoons near console | Any raccoon can initiate; first one to press E starts the hack |
| Mission ends during hack | Unhacked consoles don't count toward objective completion |
| Phase transitions mid-hack | Hack continues with original timing; no re-roll |
| Raccoon dies mid-hack | Hack continues with other raccoons still in play |
